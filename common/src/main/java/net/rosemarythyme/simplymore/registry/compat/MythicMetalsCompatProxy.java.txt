package net.rosemarythyme.simplymore.registry.compat;

import dev.architectury.registry.registries.RegistrySupplier;
import net.minecraft.item.Item;

import java.util.ArrayList;
import java.util.List;

/**  This class is NOT used to actually register these items, just to bridge the gap between the
 * Fabric module (where the items are  registered) and the Common module (where the ItemGroup is registered)
 * <p>
 * To see where these items are actually registered:
 * net.rosemarythyme.simplymore.fabric.compat.MythicMetalsCompat*/
public class MythicMetalsCompatProxy {
    public static List<RegistrySupplier<Item>> mythicMetalCompatItems = new ArrayList<>();


    public static void addToGroup(List<Item> entries) {
        mythicMetalCompatItems.forEach(item -> entries.add(item.get()));
    }
}
